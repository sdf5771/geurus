산출물 ⑤ — Agent type
타입 구분 3안 · 색을 쓰지 않는 방법
서브에이전트 타입(frontend · explore · backend 등)을 팔레트 밖 색 없이 구분해요. 세 방향을 나란히 두고 하나를 권장합니다.
// 권장: A 잎 변주 — LEAF는 EXPR과 동일한 스파스 패치 (POSE 복제 없음) const LEAF = { pair: {}, // 기본 두 잎 — 덮지 않음 triple: { "1,5":".", "1,10":".", "2,6":".", "2,9":".", "2,8":"L", "3,7":"L", "3,8":"l" }, // 가운데 잎 추가 stem: { "1,4":".", "1,5":".", "1,10":".", "1,11":".", "2,3":".", "2,4":".", "2,5":".", "2,6":".", "2,9":".", "2,10":".", "2,11":".", "2,12":".", "3,4":".", "3,5":".", "3,6":".", "3,7":".", "3,8":".", "3,9":".", "3,10":".", "3,11":"." }, // 잎 전지, 줄기만 남김 bud: { "1,4":".", "1,5":".", "1,10":".", "1,11":".", "1,7":"L", "1,8":"L", "2,3":".", "2,4":".", "2,5":".", "2,11":".", "2,12":".", "2,6":"L", "2,7":"l", "2,8":"l", "2,9":"L", "3,4":".", "3,5":".", "3,10":".", "3,11":".", "3,7":"L", "3,8":"L" }, // 봉오리 한 개 }; // 포즈별 보정 — LEAF를 포즈별로 나누지 않고 오프셋 룰로 처리한다. // LEAF는 stand 기준 좌표 하나만 유지하고, 포즈가 자기 오프셋을 선언한다. // dr = 캐노피 수직 이동 · dc = 수평 이동 · spread = 중심에서 좌우로 벌림 const LEAF_OFFSET = { stand: { dr: 0, dc: 0, spread: 0 }, squash: { dr: 1, dc: 0, spread: 2 }, // 눌려서 잎이 바깥으로 stretch: { dr: -1, dc: 0, spread: -1 }, // 늘여서 잎이 안으로 lean: { dr: 0, dc: 1, spread: 0 }, // 기울어진 줄기 쪽으로 }; // spread는 중심(열 7.5)에서 멀어지는 방향으로만 밀어 좌우 대칭을 지킨다 function leafCoord(r, c, off) { const side = c < 7.5 ? -1 : 1; return [r + off.dr, c + off.dc + side * off.spread]; } // 적용 — EXPR과 같은 모양. 표정보다 아래 레이어라 눈·입과 충돌하지 않는다. function applyLeaf(grid, poseKey, leafKey) { const patch = LEAF[leafKey]; if (!patch) return grid; const off = LEAF_OFFSET[poseKey] || LEAF_OFFSET.stand; const out = grid.map(row => row.split('')); for (const key in patch) { const [r0, c0] = key.split(',').map(Number); const [r, c] = leafCoord(r0, c0, off); if (r < 0 || r > 17 || c < 0 || c > 15) continue; out[r][c] = patch[key]; } return out.map(cells => cells.join('')); } // 조합 — 포즈 × 잎 × 표정이 독립 레이어로 쌓인다 renderSprite(applyLeaf(POSE[pose], pose, leaf), expr, fx); // 16px 미만 미니 버전은 잎 변주 생략 — 판독이 안 되고 녹색 망이 부서진다 const leafFor = (scale, type) => scale < 1 ? 'pair' : AGENT_TYPE[type].leaf; // AGENT_TYPE — 타입 → 잎 패치 키 매핑 — 새싹은 그루의 정체성이라 변주 여지가 가장 크다 const AGENT_TYPE = { frontend: { leaf: 'pair', // 기본 두 잎 — 가장 흔한 타입 badge: '◆' }, explore: { leaf: 'triple', // 세 잎이 부채처럼 펼쳐짐 badge: '◇' }, backend: { leaf: 'stem', // 잎 없이 줄기만 — 뿌리 쪽 일 badge: '▣' }, test: { leaf: 'bud', // 봉오리 한 개 badge: '◈' }, }; // 보조: 말풍선 첫 조각에 타입명을 함께 노출 — 잎만으로 헷갈릴 때의 안전장치 // "explore · Grep 중 · 8s"
산출물 ⑥ — Empty & error

산출물 ⑦ — Tray icon
트레이 아이콘 · 상태 3종
독 아이콘을 숨기므로 메뉴바가 유일한 진입점이에요. 16×16은 그루 전체를 담기엔 좁아 새싹 축약형을 씁니다 — 두 잎 + 줄기만. macOS 템플릿 이미지(검정+투명)라 라이트·다크에서 자동 반전되고, 카운트는 아이콘 옆 타이틀이 맡아요.
// 트레이 — 템플릿 이미지(알파만 사용) · 색 금지 const TRAY = { idle: { glyph: 'sprout', title: null }, // 잎 2장 · 조용함 working: { glyph: 'sprout-tall', title: '3' }, // 줄기 1px ↑ · 활동 수 blocked: { glyph: 'sprout-alert',title: '!1', blink: true }, // 잎 사이 느낌표 홈 }; // 16px: 새싹만 · 32px(@2x): 같은 글리프를 2배 정수 확대 // blocked만 1.2s 간격 깜빡임 — 나머지는 정적
